# 股票量化交易系统 - 完整软件开发计划

## 1. 项目概述

### 1.1 项目名称
**QuantTrade** - 股票量化交易系统

### 1.2 项目目标
为个人投资者和小型机构提供一套完整的A股量化交易解决方案，包括策略回测、实盘交易、风险管理和投资组合分析功能。

### 1.3 目标用户
- 个人量化投资者
- 量化交易爱好者
- 小型投资机构

### 1.4 核心功能
- 市场数据采集与存储
- 策略回测引擎
- 实盘交易执行
- 风险管理
- 投资组合管理
- WebDashboard

```python
class Stock(models.Model):
    code = models.CharField(max_length=10, unique=True)  # 600000
    name = models.CharField(max_length=50)
    industry = models.CharField(max_length=50)
    market = models.CharField(max_length=10)  # SH/SZ
    list_date = models.DateField(null=True)
    is_active = models.BooleanField(default=True)

class DailyData(models.Model):
    stock = models.ForeignKey(Stock, on_delete=models.CASCADE)
    date = models.DateField()
    open = models.DecimalField(max_digits=10, decimal_places=2)
    high = models.DecimalField(max_digits=10, decimal_places=2)
    low = models.DecimalField(max_digits=10, decimal_places=2)
    close = models.DecimalField(max_digits=10, decimal_places=2)
    volume = models.BigIntegerField()
    amount = models.DecimalField(max_digits=20, decimal_places=2)
    
    class Meta:
        unique_together = ['stock', 'date']
        indexes = [models.Index(fields=['date'])]

class MinuteData(models.Model):
```

```python
class Strategy(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    name = models.CharField(max_length=100)
    description = models.TextField()
    code = models.TextField()  # 策略Python代码

    # 状态

        choices=[('draft', '草稿'), ('active', '运行中'), ('stopped', '已停止')],
        default='draft'
    )
    
    # 时间
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

class StrategySignal(models.Model):
```

### 3.4 回测引擎模块 (backtesting)

**功能:**
- 历史回测
- 回测参数配置 (资金/手续费/滑点)
- 回测结果分析
- 绩效指标计算

**Django Models:**
```python
class Backtest(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    strategy = models.ForeignKey(Strategy, on_delete=models.CASCADE)
    
    # 回测参数
    start_date = models.DateField()
    end_date = models.DateField()
    initial_capital = models.DecimalField(max_digits=15, decimal_places=2)
    commission = models.FloatField(default=0.0003)  # 手续费率
    slippage = models.FloatField(default=0.001)  # 滑点
    
    # 结果
    final_capital = models.DecimalField(max_digits=15, decimal_places=2, null=True)
    total_return = models.FloatField(null=True)  # 总收益率
    sharpe_ratio = models.FloatField(null=True)  # 夏普比率
    max_drawdown = models.FloatField(null=True)  # 最大回撤
    win_rate = models.FloatField(null=True)  # 胜率
    
    status = models.CharField(
        choices=[('pending', '待运行'), ('running', '运行中'), ('completed', '完成'), ('failed', '失败')],
        default='pending'
    )
    
    created_at = models.DateTimeField(auto_now_add=True)
    completed_at = models.DateTimeField(null=True)

class BacktestTrade(models.Model):
    backtest = models.ForeignKey(Backtest, on_delete=models.CASCADE)
    stock = models.ForeignKey(Stock, on_delete=models.CASCADE)
    signal_type = models.CharField()
    price = models.DecimalField(max_digits=10, decimal_places=2)
    quantity = models.IntegerField()
    commission = models.DecimalField(max_digits=15, decimal_places=2)
    datetime = models.DateTimeField()
```

```python
class Order(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    strategy = models.ForeignKey(Strategy, on_delete=models.CASCADE, null=True)
    stock = models.ForeignKey(Stock, on_delete=models.CASCADE)
    
    order_type = models.CharField(choices=[('market', '市价'), ('limit', '限价')])
    direction = models.CharField(choices=[('buy', '买入'), ('sell', '卖出')])
    price = models.DecimalField(max_digits=10, decimal_places=2, null=True)
    quantity = models.IntegerField()
    
    status = models.CharField(
        choices=[('pending', '待成交'), ('partial', '部分成交'), 
                 ('filled', '全部成交'), ('cancelled', '已撤销'), ('rejected', '已拒绝')],
        default='pending'
    )
    
    broker_order_id = models.CharField(max_length=64, null=True)
    filled_quantity = models.IntegerField(default=0)
    filled_price = models.DecimalField(max_digits=10, decimal_places=2, null=True)
    
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)

class Position(models.Model):
    user = models.ForeignKey(User, on_delete=models.CASCADE)
    strategy = models.ForeignKey(Strategy, on_delete=models.CASCADE, null=True)
    stock = models.ForeignKey(Stock, on_delete=models.CASCADE)
    
    quantity = models.IntegerField()  # 持仓数量
    avg_cost = models.DecimalField(max_digits=10, decimal_places=2)  # 平均成本
    market_value = models.DecimalField(max_digits=15, decimal_places=2)  # 市值
    
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        unique_together = ['user', 'strategy', 'stock']

class Trade(models.Model):
    order = models.ForeignKey(Order, on_delete=models.CASCADE)
    filled_quantity = models.IntegerField()
    filled_price = models.DecimalField(max_digits=10, decimal_places=2)
    commission = models.DecimalField(max_digits=15, decimal_places=2)
    filled_at = models.DateTimeField(auto_now_add=True)
```
